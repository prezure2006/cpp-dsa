# Good subarrays

A good array is an array where the number of different integers in that array is exactly k.
Duplicate possible
[Two pointers , array , subarrays , map, indexing]

```

class Solution {
public:
int helper(vector<int>& good, int k){
int left = 0,res =0;
unordered_map<int, int> freq;
for (int right =0;right<good.size();right++){
    freq[good[right]]++;
    while(freq.size()>k){
        freq[good[left]]--;
        if (freq[good[left]] == 0) freq.erase(good[left]);
            left++;
    }
       res += (right - left + 1);
}
return res;
};
    int subarraysWithKDistinct(vector<int>& nums, int k) {
          return helper(nums, k) - helper(nums, k - 1);
    }
};
```
