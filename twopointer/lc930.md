[String , two pointer, prefix sum ]

# prefix sum approach

So the sum of the subarray should not go beyond goal
TC=O(n) just the size of the array we have to iterate
SC = O(1) no extra space

```
class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
        int n = nums.size();

        unordered_map<int, int> mpp;
        mpp[0] =1;



        int ans=0,s=0;
        for (int &v : nums){
            s+=v;

            int remain  = s- goal;
            if(mpp.find(remain)!= mpp.end()){
                ans+= mpp[remain];

            }
               mpp[s]++;
        }


        return ans;
    }
};
```

# two pointer approach

TC = O(n) SC = O(1)

```
class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
     int l =0, r=0;
     int sum =0;
     int ans=0;
     int count_zero = 0;
int n = nums.size();
     while(r<n){
        sum +=nums[r];
       while(l<r && (nums[l] == 0 || sum >goal)){
        if(nums[l]==0){
            count_zero++;
        }else{
            count_zero=0;
        }
            sum = sum - nums[l];
            l++;
        }
         if(sum == goal){
            ans = ans +1 + count_zero;
        }
        r++;
     }
     return ans;
    }
};
```
