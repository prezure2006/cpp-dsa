class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
     int l =0, r=0;
     int sum =0;
     int ans=0;
     int count_zero = 0;
int n = nums.size();
     while(r<n){
        sum +=nums[r];
       while(l<r && (nums[l] == 0 || sum >goal)){
        if(nums[l]==0){
            count_zero++;
        }else{
            count_zero=0;
        }
            sum = sum - nums[l];
            l++;
        }
         if(sum == goal){
            ans = ans +1 + count_zero;
        }
        r++;
     }
     return ans;
    }
};
